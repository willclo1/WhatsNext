# Runtime dependencies for the Six Degrees API.
#
# Direct dependencies only — transitive ones (pydantic, starlette, anyio…)
# are resolved by pip. Regenerate pins with:
#   ./venv/bin/pip list --format=freeze
#
# Deliberately NOT included, though present in the local venv:
#   dotenv          — a stub package that just re-exports python-dotenv;
#                     installing both shadows the real one.
#   fastapi-cli,
#   fastapi-cloud-cli,
#   sentry-sdk,
#   httpx, typer…   — developer tooling pulled in by `fastapi[standard]`,
#                     not used by the API at runtime.

fastapi==0.141.1

# [standard] adds uvloop + httptools, which is what you want serving this.
uvicorn[standard]==0.52.1

SQLAlchemy==2.0.51

# [binary] ships a self-contained libpq wheel. Without it psycopg falls back to
# the pure-Python implementation and needs libpq present on the host — which a
# slim container does not have.
psycopg[binary]==3.3.4

python-dotenv==1.2.2
